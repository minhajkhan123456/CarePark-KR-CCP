        % EV bays have chargers

        % Staff badge is valid today

        % A security ban bans the person

  t1  [s0] on_shift(P) :- shift(P,S,E), drives(P,V), arrival(V,T), sub(S,30,S0), ge(T,S0), le(T,E).
        % On shift: from 30 min before start to shift end
  t2  [s0] in_appointment_window(P) :- appointment(P,_,T0), drives(P,V), arrival(V,T), sub(T0,60,A), add(T0,30,B), ge(T,A), le(T,B).
        % Appointment window: 60 min before to 30 min after

  t4  [s0] delivery_window(P) :- delivery_slot(P,A,B), drives(P,V), arrival(V,T), ge(T,A), le(T,B).

        % Patients not banned are cleared

        % Visitors not banned are cleared
  a4  [s1] cleared(P) :- supplier(P), delivery_window(P), not banned(P).

  a5  [s1] admitted(V) :- emergency_vehicle(V).

Integrity constraints
  ic1 :- alloc(V1,B,F1,T1), alloc(V2,B,F2,T2), V1 \= V2, F1 < T2, F2 < T1.
      one bay, two vehicles at the same time
  ic2 :- alloc(V,B1,_,_), alloc(V,B2,_,_), B1 \= B2.
      one vehicle, two bays at the same time
  ic3 :- alloc(V,B,_,_), height(V,H), bay_zone(B,Z), zone_level(Z,L), route_clearance(L,C), H > C.
      vehicle taller than its route clearance
  ic4 :- alloc(V,B,_,_), vclass(V,C), bay_type(B,T), not class_fits(C,T).
      vehicle class cannot use the bay type
  ic5 :- drives(P1,V), drives(P2,V), P1 \= P2.
      vehicle driven by two people
  ic6 :- vclass(V,C1), vclass(V,C2), C1 \= C2.
      vehicle with two classes
  ic7 :- bay_zone(B,Z1), bay_zone(B,Z2), Z1 \= Z2.
      bay in two zones
  ic8 :- shift(P,S,E), E =< S.
      shift ends before it starts
  ic9 :- staff_badge(P,I,E), E < I.
      badge expires before it is issued
  ic10 :- appointment(P,_,_), not patient(P).
      appointment for someone who is not a patient
  ic11 :- vclass(V,ambulance), drives(P,V), not paramedic(P).
      ambulance not driven by a paramedic
  ic12 :- occupies(V,B), out_of_service(B).
      vehicle parked in a bay that is out of service
  ic13 :- staff_member(P), visitor(P).
      staff role and visitor role on one person
